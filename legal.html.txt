<script>
    function checkJurisdiction() {
        let amt = parseFloat(document.getElementById('claimAmount').value);
        if (!amt) return;
        let res = "";
        if (amt <= 37900) {
            res = "בית משפט לתביעות קטנות (או בית משפט השלום)";
        } else if (amt <= 2500000) {
            res = "בית משפט השלום";
        } else {
            res = "בית המשפט המחוזי";
        }
        document.getElementById('jurisResult').innerText = "הערכאה המוסמכת: " + res;
    }

    function calcCourtFee() {
        let amt = parseFloat(document.getElementById('feeClaim').value);
        let type = document.getElementById('courtType').value;
        if (!amt) return;
        let fee = 0;
        if (type === 'small') {
            fee = Math.max(amt * 0.01, 50); // לפחות 50 ש"ח
        } else {
            fee = amt * 0.025;
        }
        document.getElementById('feeResult').innerText = `אגרה משוערת: ${fee.toLocaleString()} ₪ (מחצית ראשונה 1.25% בהגשה)`;
    }
</script>
